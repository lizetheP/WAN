!---------------------------------------------------------------
!Router A
!---------------------------------------------------------------
en
config t

hostname RA

interface G0/1
ip address 10.0.1.254 255.255.254.0
ip helper-address 10.0.3.129
no shut

! Para resolver el problema de que el router A no tiene DHCP, 
! pero alguien más si lo tiene y lo voy a poder contactar
! por algunas de sus tres interfaces del router B.
! Con esto el router A contacta al router B para que le entregue
! direcciones IP dinámicas a todos los equipos de la subred.

interface Serial0/1/0
ip address 10.0.3.130 255.255.255.252
clock rate 128000
no shut

router ospf 1
passive-interface GigabitEthernet0/1
network 10.0.0.0 0.0.1.255 area 0
network 10.0.3.128 0.0.0.3 area 0

!---------------------------------------------------------------
!Router B
!---------------------------------------------------------------
en
config t

hostname RB

interface G0/1
ip address 10.0.7.254 255.255.252.0
no shut
exit

interface Serial0/1/0
ip address 65.10.8.66 255.255.255.224
no shut
exit

interface Serial0/1/1
ip address 10.0.3.129 255.255.255.252
no shut
exit

!DHCP
!----------------
! DIRECCIONES EXCLUIDAS DE LAS INTERFACES PUERTAS DE ENLACE PREDERTMINADAS

ip dhcp excluded-address 10.0.7.254
ip dhcp excluded-address 10.0.1.253 10.0.1.254

ip dhcp pool RA_G0/1
network 10.0.0.0 255.255.254.0
default-router 10.0.1.254
dns-server 5.5.5.5

ip dhcp pool RB_G0/1
network 10.0.4.0 255.255.252.0 
default-router 10.0.7.254
dns-server 5.5.5.5

! 1024 direcciones disponibles - 2 y una ya está ocupada por el gateway.

ip route 0.0.0.0 0.0.0.0 s0/1/0

router ospf 1
passive-interface GigabitEthernet0/1
passive-interface Serial0/1/0
network 10.0.3.128 0.0.0.3 area 0
network 10.0.4.0 0.0.3.255 area 0
default-information originate

!NAT
!----------------------------------------------------------------
!Traduccion dinamica uno a uno, por cada direccion IP privada 
!se asignara una direccion IP publica.

!Definir un pool de direcciones globales (públicas) que serán asignadas cuando sean necesarias.
ip nat pool MyPool 170.15.9.33 170.15.9.63 netmask 255.255.255.224

!Definir una ACL estándar con las direcciones IP privadas que tienen permiso a ser traducidas
access-list 10 permit any
!access-list 10 permit 10.0.0.0 0.0.3.255

!Establecer la traducción de direcciones utilizando una lista de acceso:
ip nat inside source list 10 pool MyPool

255.255.254.0     11111110
255.255.252.0     11111100
255.255.255.252   11111111
----------------
255.255.252.0

!PAT
!----------------------------------------------------------------
!Traduccion dinamica de muchas direcciones IPs privadas contra 
!una o pocas direcciones IP públicas para salir al exterior.

ip nat inside source list 10 pool MyPool overload

! Traduccion estatica
!--------------------------------
ip nat inside source static 10.0.1.253 170.15.9.32

int g0/1
ip nat inside
exit

int s0/1/1
ip nat inside
exit

int s0/1/0
ip nat outside
exit
